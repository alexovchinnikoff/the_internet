# Проект автоматизации тестирования (Requests + Playwright)

## Структура проекта

project/  
├── README.md  # Описание проекта, инструкции по установке и использованию.  
├── requirements.txt  # Список зависимостей для проекта.  
├── pytest.ini  # Конфигурация pytest для запуска тестов.  
├── .env  # Файл с переменными окружения (например, API ключи).  
├── .gitignore  # Файлы и директории, которые не нужно включать в репозиторий.  
├── config/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── settings.py  # Конфигурационные настройки проекта.  
│ └── urls.py  # Список URL-адресов, используемых в приложении.  
├── core/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── api_client.py  # Класс для работы с API.  
│ ├── browser_manager.py  # Класс для управления браузером.  
│ └── logger.py  # Класс для логирования.  
├── models/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── request_models.py  # Модели для запросов к API.  
│ └── response_models.py  # Модели для ответов от API.  
├── fixtures/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── api_fixtures.py  # Фикстуры для тестов API.  
│ └── ui_fixtures.py  # Фикстуры для тестов пользовательского интерфейса (UI).  
├── pages/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── base_page.py  # Базовый класс для страниц.  
│ ├── login_page.py  # Класс для страницы входа.  
│ ├── main_page.py  # Класс для главной страницы.  
│ └── components/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ └── header.py  # Класс для заголовка страниц.  
├── api/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── endpoints/  
│ │ ├── __init__.py  # Мета-файл для Python пакета.  
│ │ ├── auth_endpoints.py  # Эндпоинты для аутентификации.  
│ │ └── user_endpoints.py  # Эндпоинты для работы с пользователями.  
│ └── services/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ └── api_service.py  # Сервисы для работы с API.  
├── tests/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── api/  
│ │ ├── __init__.py  # Мета-файл для Python пакета.  
│ │ ├── test_auth.py  # Тесты аутентификации API.  
│ │ └── test_users.py  # Тесты работы с пользователями API.  
│ └── ui/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── test_login.py  # Тесты страницы входа.  
│ └── test_navigation.py  # Тесты навигации по приложению.  
├── utils/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── helpers.py  # Утилитные функции.  
│ ├── data_generators.py  # Генераторы тестовых данных.  
│ └── validators.py  # Валидаторы данных.  
├── data/  
│ ├── __init__.py  # Мета-файл для Python пакета.  
│ ├── test_data.json  # Тестовые данные в формате JSON.  
│ └── credentials.json  # Учетные записи пользователей в формате JSON.  
└── reports/  
    └── (автоматически создается)  # Директория для автоматически создаваемых отчетов о тестировании.  
└── screenshots/  
    └── (автоматически создается)  # Директория для скриншотов, сделанных во время выполнения тестов.
